const RELATIVE = new Intl.RelativeTimeFormat('en', { numeric: 'auto' });

export function timeAgo(iso: string): string {
  const diff = Date.now() - new Date(iso).getTime();
  const mins = Math.round(diff / 60_000);
  if (mins < 1) return 'just now';
  if (mins < 60) return RELATIVE.format(-mins, 'minute');
  const hours = Math.round(mins / 60);
  if (hours < 24) return RELATIVE.format(-hours, 'hour');
  return RELATIVE.format(-Math.round(hours / 24), 'day');
}

/** Avatar initials from a label ("Mary Smith" â†’ "MS", fallback "VI"). */
function initialsOf(label: string): string {
  const parts = label.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return 'VI';
  return parts
    .map((p) => p[0])
    .slice(0, 2)
    .join('')
    .toUpperCase();
}

/** Shared inbox conversation list: search, status/assignee filters, live updates. */
export default function InboxList({ selectedId }: Props) {
  const { user } = useAuth();
  const [items, setItems] = useState<Conversation[]>([]);
  const [cursor, setCursor] = useState<string | null>(null);
  const [status, setStatus] = useState<ConversationStatus | ''>('');
  const [assignee, setAssignee] = useState<'me' | 'unassigned' | 'any'>('any');
  const [query, setQuery] = useState('');
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const socketRef = useRef<Socket | null>(null);

  const load = useCallback(
    async (nextCursor?: string) => {
      setError('');
      try {
        const page = await api.listConversations({
          ...(status ? { status } : {}),
          assignee,
          ...(nextCursor ? { cursor: nextCursor } : {}),
          limit: 50,
        });
        setItems((prev) => (nextCursor ? [...prev, ...page.items] : page.items));
        setCursor(page.nextCursor);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Failed to load conversations');
      } finally {
        setLoading(false);
      }
    },
    [status, assignee],
  );

  useEffect(() => {
    setLoading(true);
    void load();
  }, [load]);

  // Live inbox updates: workspace room events (assign/close/new) â†’ refetch.
  useEffect(() => {
    const token = getAccessToken();
    if (!token) return;
    const socket = connectAgentSocket(token, {
      onConversationUpdated: () => void load(),
    });
    socketRef.current = socket;
    return () => {
      socket.close();
      socketRef.current = null;
    };
  }, [load]);

  // Global search: initial ?q= from the topbar jump + live "lc-search" events.
  useEffect(() => {
    const initial = new URLSearchParams(window.location.search).get('q');
    if (initial) setQuery(initial);
    const onSearch = (e: Event) => setQuery(String((e as CustomEvent<string>).detail ?? ''));
    window.addEventListener('lc-search', onSearch);
    return () => window.removeEventListener('lc-search', onSearch);
  }, []);

  const needle = query.trim().toLowerCase();
  const visible = needle
    ? items.filter((c) =>
        `${c.subject ?? ''} ${c.visitor?.name ?? ''} ${c.visitor?.email ?? ''} ${c.visitorId ?? ''}`.toLowerCase().includes(needle),
      )
    : items;

  return (
    <div className="inbox-list">
      <div className="list-tools">
        <label className="search-pill">
          <SearchIcon size={15} />
          <input
            placeholder="Search conversations"
            aria-label="Search conversations"
            value={query}
            onChange={(e) => setQuery(e.target.value)}
          />
        </label>
        <select
          className="pill-select"
          aria-label="Status"
          value={status}
          onChange={(e) => setStatus(e.target.value as ConversationStatus | '')}
        >
          <option value="">All statuses</option>
          <option value="open">Open</option>
          <option value="pending">Pending</option>
          <option value="closed">Closed</option>
        </select>
      </div>
      <div className="list-tools sub">
        <select
          className="pill-select"
          aria-label="Assignee"
          value={assignee}
          onChange={(e) => setAssignee(e.target.value as typeof assignee)}
        >
          <option value="any">Anyone</option>
          <option value="me">Assigned to me</option>
          <option value="unassigned">Unassigned</option>
        </select>
        <span className="list-count">
          {visible.length} {visible.length === 1 ? 'conversation' : 'conversations'}
        </span>
        <button
          className="icon-btn"
          onClick={() => void load()}
          aria-label="Refresh conversations"
          data-tip="Refresh"
        >
          <RefreshIcon size={15} />
        </button>
      </div>

      {error && <div className="alert error" style={{ margin: 12 }}>{error}</div>}
      {loading && items.length === 0 && <div className="loading">Loadingâ€¦</div>}
      {!loading && visible.length === 0 && !error && (
        <div className="empty-state" style={{ padding: 40 }}>
          {needle ? 'No conversations match your search.' : 'No conversations yet.'}
        </div>
      )}
      {visible.map((c) => {
        const label =
          c.subject ??
          c.visitor?.name ??
          c.visitor?.email ??
          `Visitor ${c.visitorId?.slice(0, 8) ?? 'â€”'}`;
        const assigned = Boolean(c.assignedTo);
        const mine = Boolean(c.assignedTo) && c.assignedTo === user?.id;
        return (
          <Link
            key={c.id}
            className={`conv-item${selectedId === c.id ? ' active' : ''}`}
            href={`/conversations/${c.id}`}
            aria-current={selectedId === c.id ? 'page' : undefined}
          >
            <span className="conv-avatar" aria-hidden="true">
              {initialsOf(label)}
            </span>
            <span className="conv-main">
              <span className="conv-row">

    </div>
  );
}

const EMOJIS = ['\uD83D\uDC4D', '\uD83C\uDF89', '\uD83D\uDE04', '\uD83D\uDE4F', '\u2705', '\uD83D\uDD25', '\u2764\uFE0F', '\uD83D\uDE05'];

function whenLabel(iso: string): string {
  const d = new Date(iso);
  const sameDay = d.toDateString() === new Date().toDateString();
  const day = sameDay
    ? 'Today'
    : d.toLocaleDateString('en', { month: 'short', day: 'numeric' });
  return `${day} ${TIME.format(d)}`;
}

/** Conversation detail: inbox list + thread (channel tabs, bubbles, composer)
 *  + right-hand contact panel (status, visitor details, sections, actions). */
export default function ConversationDetailPage() {
  const params = useParams<{ id: string }>();
  const conversationId = params.id;
  const { user } = useAuth();

  const [conversation, setConversation] = useState<Conversation | null>(null);
  const [visitor, setVisitor] = useState<VisitorDetail | null>(null);
  const [messages, setMessages] = useState<Message[]>([]);
  const [draft, setDraft] = useState('');
  const [subject, setSubject] = useState('');
  const [error, setError] = useState('');
  const [typing, setTyping] = useState(false);
  const [online, setOnline] = useState(false);
  const [emojiOpen, setEmojiOpen] = useState(false);

  const seenIds = useRef(new Set<string>());
  const messagesEndRef = useRef<HTMLDivElement | null>(null);
  const composerRef = useRef<HTMLTextAreaElement | null>(null);
  const socketRef = useRef<Socket | null>(null);
  const typingTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  const appendMessage = useCallback((msg: Message) => {
    if (seenIds.current.has(msg.id)) return;
    seenIds.current.add(msg.id);
    setMessages((prev) => [...prev, msg]);
  }, []);

  // Load conversation + first page of history.
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const [conv, page] = await Promise.all([
          api.getConversation(conversationId),
          api.listMessages(conversationId),
        ]);
        if (cancelled) return;
        setConversation(conv);
        setSubject(conv.subject ?? '');
        for (const m of page.items) seenIds.current.add(m.id);
        setMessages(page.items);
      } catch (err) {
        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load');
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [conversationId]);

  // Contact panel: resolve the visitor behind this conversation.
  useEffect(() => {
    const id = conversation?.visitorId;
    if (!id) return;
    let cancelled = false;
    api
      .getVisitor(id)
      .then((v) => {
        if (!cancelled) setVisitor(v);
      })
      .catch(() => undefined);
    return () => {
      cancelled = true;
    };
  }, [conversation?.visitorId]);

  // Realtime: visitor messages, typing, presence, conversation updates.
  useEffect(() => {
    const token = getAccessToken();
    if (!token) return;
    const socket = connectAgentSocket(token, {
      onMessage: (payload) => {
        const msg = payload as Message & { conversationId: string };
        if (msg.conversationId === conversationId) appendMessage(msg);
      },
      onTyping: (payload) => {
        const p = payload as { conversationId: string; typing: boolean };
        if (p.conversationId !== conversationId) return;
        setTyping(p.typing);
        if (typingTimer.current) clearTimeout(typingTimer.current);
        typingTimer.current = setTimeout(() => setTyping(false), 3000);
      },
      onVisitorPresence: (payload) => {
        const p = payload as { visitorId: string; online: boolean };
        if (p.visitorId === conversation?.visitorId) setOnline(p.online);
      },
      onConversationUpdated: (payload) => {
        const p = payload as { conversationId?: string };
        if (p.conversationId === conversationId) {
          void api
            .getConversation(conversationId)
            .then(setConversation)
            .catch(() => undefined);
        }
      },
    });
    socketRef.current = socket;
    if (conversation) joinConversation(socket, conversationId);
    return () => {
      socket.close();
      socketRef.current = null;
    };
  }, [conversationId, conversation, appendMessage]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ block: 'end' });
  }, [messages.length, typing]);

  async function onSend(e: React.FormEvent) {
    e.preventDefault();
    const body = draft.trim();
    if (!body) return;
    setDraft('');

      appendMessage(await api.sendMessage(conversationId, body));
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Send failed');
    }
  }

  async function assignToMe() {
    if (!user) return;
    try {
      setConversation(await api.updateConversation(conversationId, { assignedTo: user.id }));
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Assign failed');
    }
  }

      setError(err instanceof Error ? err.message : 'Update failed');
    }
  }

    if (!conversation) return;
    const next = subject.trim();
    if (next === (conversation.subject ?? '')) return;
    try {
      setConversation(await api.updateConversation(conversationId, { subject: next || null }));
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Subject update failed');
      setSubject(conversation.subject ?? '');
    }
  }

  function insertEmoji(e: string) {
    setEmojiOpen(false);
    setDraft((d) => d + e);
    composerRef.current?.focus();
  }

    visitor?.name ??
    (conversation?.visitorId ? `Visitor ${conversation.visitorId.slice(0, 8)}` : 'Visitor');
  const [first = '', ...restParts] = displayName.split(' ');
  const last = restParts.join(' ') || '—';
  const metaEntries = Object.entries(visitor?.meta ?? {});

  return (
    <div className="inbox with-contact">
      <InboxList selectedId={conversationId} />

      <div className="inbox-detail thread">
        <div className="thread-head">
          <Link
            href="/conversations"
            className="icon-btn"
            aria-label="Back to inbox"
            data-tip="Back to inbox"
          >
            <ArrowLeftIcon size={17} />
          </Link>
          <span
            className={`status-dot${online ? ' online' : ''}`}
            title={online ? 'Visitor online' : 'Visitor offline'}
          />
          <span className="title">{conversation?.subject ?? displayName}</span>
          {conversation && (
            <span className={`badge ${conversation.status}`}>{conversation.status}</span>
          )}
        </div>

        <div className="tabs" role="tablist" aria-label="Channels">
          <button className="tab active" type="button" role="tab" aria-selected="true">
            All
          </button>
          <button
            className="tab"
            type="button"
            role="tab"
            aria-selected="false"
            disabled
            title="Channel not connected"
          >
            SMS
          </button>
          <button
            className="tab"
            type="button"
            role="tab"

            disabled

            Email

const RELATIVE = new Intl.RelativeTimeFormat('en', { numeric: 'auto' });

function timeAgo(iso: string): string {
  const diff = Date.now() - new Date(iso).getTime();
  const mins = Math.round(diff / 60_000);
  if (mins < 1) return 'just now';
  if (mins < 60) return RELATIVE.format(-mins, 'minute');
  const hours = Math.round(mins / 60);
  if (hours < 24) return RELATIVE.format(-hours, 'hour');
  return RELATIVE.format(-Math.round(hours / 24), 'day');
}
