40	function titleFor(pathname: string): string {
41	  if (pathname.startsWith('/conversations/')) return 'Conversation';
42	  if (pathname.startsWith('/visitors/')) return 'Visitor';
43	  if (pathname.startsWith('/settings/api-keys')) return 'API Keys';
44	  if (pathname.startsWith('/settings/widget')) return 'Widget Settings';
45	  if (pathname.startsWith('/settings/members')) return 'Team';
46	  if (pathname.startsWith('/settings/security')) return 'Security';
47	  if (pathname.startsWith('/settings/activity')) return 'Activity';
48	  if (pathname.startsWith('/settings')) return 'Settings';
49	  if (pathname.startsWith('/visitors')) return 'Visitors';
50	  return 'Conversations';
51	}
